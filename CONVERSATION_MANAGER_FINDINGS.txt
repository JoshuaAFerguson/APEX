      // Remove oldest messages if over limit
      while (this.context.messages.length > this.maxContextMessages) {
        this.context.messages.shift();

      let estimatedTokens = this.estimateTokens();

      // Consistent minimum message logic
      const minMessages = this.totalMessagesAdded < 50 ? 10 : 2;

      while (estimatedTokens > this.maxContextTokens &&
             this.context.messages.length > minMessages) {
        const removed = this.context.messages.shift();
        if (removed) {

          Math.ceil(JSON.stringify(m.metadata).length / 4) : 0;

        return sum + contentTokens + metadataTokens + structureTokens;
